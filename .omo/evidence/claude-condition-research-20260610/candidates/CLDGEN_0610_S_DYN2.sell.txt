# 시총별 동적 청산 v2 — 윈도우 함수 0개(엔진 계산예산 대응), 스칼라 전용
매도 = False

if 시가총액 < 5000:
    손절률 = -0.8
    익절률 = 2.0
    트레일링시작 = 1.0
    트레일링폭 = 0.45
    최대보유시간 = 150
else:
    손절률 = -1.0
    익절률 = 2.4
    트레일링시작 = 1.3
    트레일링폭 = 0.6
    최대보유시간 = 240

if 등락율 > 29.5:
    매도 = True
elif 시분초 >= 92800:
    매도 = True
elif 수익률 <= 손절률:
    매도 = True
elif 수익률 >= 익절률:
    매도 = True
elif 최고수익률 >= 트레일링시작 and 수익률 <= 최고수익률 - 트레일링폭:
    매도 = True
elif 수익률 < 0 and 초당매도수량 > 초당매수수량 * 1.5 and 현재가 < 현재가N(1):
    매도 = True
elif 보유시간 >= 최대보유시간:
    매도 = True

if 매도:
    self.Sell()
