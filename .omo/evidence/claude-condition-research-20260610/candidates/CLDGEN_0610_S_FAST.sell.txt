# 시초 스캘프용 빠른 청산: 얕은 손절 + 트레일링 + 짧은 시간 청산
매도 = False

if 등락율 > 29.5:
    매도 = True
elif 시분초 >= 92800:
    매도 = True
elif 수익률 <= -0.8:
    매도 = True
elif 수익률 >= 2.0:
    매도 = True
elif 최고수익률 >= 0.8 and 수익률 <= 최고수익률 - 0.5:
    매도 = True
elif 데이터길이 >= 11 and 수익률 < 0 and 체결강도 < 체결강도평균(10, 1) * 0.9 and 초당매도수량 > 초당매수수량:
    매도 = True
elif 보유시간 >= 90:
    매도 = True

if 매도:
    self.Sell()
