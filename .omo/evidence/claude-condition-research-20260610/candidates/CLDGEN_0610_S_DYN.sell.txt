# v5.0 리포트 기반: 시가총액별 동적 청산 + 세력이탈 점수(단순화)
매도 = False

if 시가총액 < 5000:
    손절률 = -0.8
    익절률 = 2.0
    트레일링시작 = 1.0
    트레일링폭 = 0.45
    최대보유시간 = 150
else:
    손절률 = -1.0
    익절률 = 2.4
    트레일링시작 = 1.3
    트레일링폭 = 0.6
    최대보유시간 = 240

이탈점수 = 0
if 데이터길이 >= 11 and 체결강도 < 체결강도평균(10, 1) * 0.85:
    이탈점수 += 2
if 초당매도수량 > 초당매수수량 * 1.5:
    이탈점수 += 1
if 데이터길이 >= 11 and 누적초당매도수량(10) > 누적초당매수수량(10) * 1.3:
    이탈점수 += 2
if 최고수익률 >= 1.0 and 고가미갱신지속틱수() >= 15:
    이탈점수 += 2
if 데이터길이 >= 2 and 매수총잔량 < 매수총잔량N(1) * 0.7:
    이탈점수 += 1

if 등락율 > 29.5:
    매도 = True
elif 시분초 >= 92800:
    매도 = True
elif 수익률 <= 손절률:
    매도 = True
elif 수익률 >= 익절률:
    매도 = True
elif 최고수익률 >= 트레일링시작 and 수익률 <= 최고수익률 - 트레일링폭:
    매도 = True
elif 이탈점수 >= 4:
    매도 = True
elif 수익률 < 0 and 이탈점수 >= 3:
    매도 = True
elif 보유시간 >= 최대보유시간:
    매도 = True

if 매도:
    self.Sell()
